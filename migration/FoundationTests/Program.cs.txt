static void Require(bool valid, string message) { if (!valid) throw new Exception(message); }

    var value = PanelBudget.value(card.cost, card.attack, card.defense);
    Require(value >= bounds.low && value <= bounds.high, "panel budget " + card.name);
}
using var oracle = JsonDocument.Parse(File.ReadAllBytes(Path.Combine(root, "migration/FoundationTests/oracle.json")));
foreach (var row in oracle.RootElement.GetProperty("rng").EnumerateArray())
{
    var rng = new PythonRandom(BigInteger.Parse(row.GetProperty("seed").GetString()!));
    foreach (var expected in row.GetProperty("random").EnumerateArray()) Require(BitConverter.DoubleToInt64Bits(rng.random()) == BitConverter.DoubleToInt64Bits(expected.GetDouble()), "random " + row.GetProperty("seed"));
    foreach (var item in row.GetProperty("bits").EnumerateArray()) Require(rng.getrandbits(item.GetProperty("k").GetInt32()).ToString() == item.GetProperty("value").GetString(), "randbits");
    foreach (var item in row.GetProperty("ranges").EnumerateArray()) Require(rng.randrange(item.GetProperty("n").GetInt32()) == item.GetProperty("value").GetInt32(), "randrange");
    var shuffled = Enumerable.Range(0, 40).ToList(); rng.shuffle(shuffled);
    Require(shuffled.SequenceEqual(row.GetProperty("shuffle").EnumerateArray().Select(x => x.GetInt32())), "shuffle");
    foreach (var item in row.GetProperty("samples").EnumerateArray())
    {
        var population = Enumerable.Range(0, item.GetProperty("n").GetInt32()).ToArray();
        Require(rng.sample(population, item.GetProperty("k").GetInt32()).SequenceEqual(item.GetProperty("value").EnumerateArray().Select(x => x.GetInt32())), "sample");
    }
    Require(rng.choices(new[] { "a", "b", "c", "d" }, new double[] { 0, 0.1, 0.4, 0.5 }, 30).SequenceEqual(row.GetProperty("choices").EnumerateArray().Select(x => x.GetString()!)), "choices");
    var clone = rng.Clone(); for (var i = 0; i < 40; i++) Require(rng.getrandbits(64) == clone.getrandbits(64), "rng clone");
    var longRng = new PythonRandom(BigInteger.Parse(row.GetProperty("seed").GetString()!));
    var bytes = new byte[20_000 * 4];
    for (var i = 0; i < 20_000; i++) BinaryPrimitives.WriteUInt32LittleEndian(bytes.AsSpan(i * 4, 4), longRng.NextUInt32());
    Require(Convert.ToHexString(SHA256.HashData(bytes)).ToLowerInvariant() == row.GetProperty("long_words_sha256").GetString(), "MT19937 repeated twists");
}

    Require((a+b).ToString() == item.GetProperty("add").GetString(), "fraction add");
    Require((a*b).ToString() == item.GetProperty("mul").GetString(), "fraction multiply");
    Require((a/b).ToString() == item.GetProperty("div").GetString(), "fraction divide");
    Require((a%b).ToString() == item.GetProperty("mod").GetString(), "fraction modulo");
    Require(a.Floor().ToString() == item.GetProperty("floor").GetString(), "fraction floor");
    Require(a.Ceiling().ToString() == item.GetProperty("ceil").GetString(), "fraction ceil");
    Require(a.Pow(-3).ToString() == item.GetProperty("pow").GetString(), "fraction power");
}
